\begin{definicion}[Población]
  Una población de estudio es un conjunto de ejemplares a estudiar.
\end{definicion}

\begin{definicion}[Espacio de probabilidad]
    $(\Omega, \mathcal{F}, P)$ es un espacio de probabilidad si
    \begin{enumerate}
      \item $\Omega$ es un conjunto no vacío
      \item $\mathcal{F}$ es una $\sigma$ álgebra sobre $2^\Omega = \{U | U \subseteq \Omega \}$
      \item $P: \mathcal{F} \mapsto [0, 1]$ es una medida de probabilidad (medida tal que $P(\mathcal F) = 1$)
    \end{enumerate}
\end{definicion}

\begin{definicion}[Variable aleatoria]
  Se dice que $X:\Omega \mapsto E$ es una variable aleatoria si:
  \begin{enumerate}
    \item $(\Omega, \mathcal{F}, P)$ es un espacio de probabilidad
    \item $E$ es un espacio medible (típicamente $\R^n$)
    \item $X$ es una función medible
  \end{enumerate}
\end{definicion}

\begin{definicion}[Muestra]
  Una muestra de una población es una $n$-tupla de elementos de la población $P$.
  la distribución de $X$ para cada $i$.
\end{definicion}

\begin{ejemplo}
  \begin{enumerate}
    \item Altura de una persona elegida al azar, medida en cm: $\{175\}$.
    \item Altura y peso de una persona, medidas en cm y kilogramos: $\{(175, 75)\}$.
    \item Altura y peso de dos personas elegidas al azar con reemplazo, medida en cm y kilogramos:
      $\{(175, 75), (168, 66)\}$.
  \end{enumerate}
\end{ejemplo}

\begin{definicion}[Muestra Aleatoria Simple con reemplazamiento]
  Una muestra $M$ donde los elementos del vector son independientes:
  $(X_1, ..., X_n)$ tales que

  \begin{displaymath}
  \begin{cases}
  X_i \perp X_j \;\forall j \neq i\\
  X_i \sim X \;\forall i = 1, ..., n
  \end{cases}
  \end{displaymath}

  donde el símbolo $\perp$ significa independencia.

  Esto equivale a un vector de $n$ elementos de $P$ elegidos al azar
  con distribución uniforme sucesivamente y con reemplazamiento,
  justificando el nombre.
\end{definicion}
\begin{corolario}

	P(x_1, ..., x_n) = \prod_{i=1}^n P(x_i)
	$$
\end{corolario}

\begin{definicion}[Muestra Aleatoria simple sin reemplazamiento]
% TODO: source cite cita
  Una muestra $M$ donde los elementos de la población se eligen de forma que
  $$ \forall k \leq n, P(S_1) = P(S_2) \forall S_1, S_2 \in \{U \in 2^P | \#(U) = k \},  $$

  Donde $\#$ denota la función que cuenta el número de elementos de un conjunto.
  Esto es, la probabilidad de elegir cualquier subconjunto subconjunto con $k$ elementos es la misma
  que la de cualquier otro subconjunto de $k$ elementos.
\end{definicion}

\begin{corolario}
  Una muestra aleatoria simple sin reemplazo no es independiente, puesto que
  no se puede elegir el mismo sujeto de la población dos veces.

\begin{definicion}[Grafo]
    Un grafo es un par $G = (V, E)$ donde $V$ es una colección de elementos llamados vértices
    y $E$ es una $2$-tupla de elementos de $V$ llamadas aristas.
    Si consideramos las $2$-tuplas de $E$ ordenadas, se le llama un grafo dirigido y las aristas son dirigidas.
    Si no las consideramos ordenadas, los elementos de $E$ son conjuntos y el grafo es no dirigido.
\end{definicion}

\begin{definicion}[Camino]
    Un camino en un grafo $G = (V, E)$ es una $n$-tupla $(c_i)_{i = 1, ..., n}$ de elementos de $V$ tales que
    $\forall i \in \{2, ..., n\} \exists e_i = (e_i^1, e_i^2) \in E$ cumpliendo $e_i^1 = c_{i - 1}$ y $e_i^2 = c_i$
\end{definicion}

\begin{definicion}[Ciclo]
    Un ciclo en un grafo es un camino $(c_i)_{i = 2, ..., n}$ tal que $c_1 = c_n$.
\end{definicion}

\begin{definicion}[Modelo estadístico]\cite[1.2]{mci}
    % TODO: cite source
    Un modelo estadístico es un par $(\chi, \mathcal{P})$, donde
    $\chi$ es el espacio muestral y
    $\mathcal{P}$ es una familia de distribuciones de probabilidad sobre $\chi$.

    Típicamente diremos que una variable aleatoria pertecene a un modelo obviando
    el espacio muestral de dicha variable.
\end{definicion}

\begin{nota}
	Típicamente diremos que $\mathcal{P} \in \mathcal{M}$, obviando el espacio muestral.
\end{nota}

\begin{observacion}
    Un modelo estadístico representa una colección de asunciones respecto a la generación de una muestra.
\end{observacion}
