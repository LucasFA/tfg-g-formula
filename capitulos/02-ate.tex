\chapter{Framework de Potential Outcomes}

\begin{definicion}[Modelo estadístico]\cite[1.2]{mci}\cite[1.3.1]{targetedLearning}\label{def:modelo}
	Un modelo $\mathcal{M}$ es una
	familia de distribuciones de probabilidad.
\end{definicion}

\begin{notac}
	Denotaremos $P \in \mathcal{M}$ para indicar $P \in \mathcal{P}$, obviando el espacio muestral.
\end{notac}
\begin{observacion}
	Un modelo representa una colección de asunciones respecto a la generación de muestras.
\end{observacion}

\section{Introducción}
Basándonos en \cite[capítulo 2]{targetedLearning}, consideremos el modelo estadístico $\mathcal{M} \ni P$,
donde $P$ es la distribución conjunta del triplete $(X, A, Y)$
de variables aleatorias observables, y supongamos que $Y$ es una variable aleatoria binaria.

A $X$ lo llamamos las covariantes, a $A$ el tratamiento, y a $Y$ el resultado (\textit{outcome}, del inglés).

Consideramos entonces un modelo (familia de distribuciones) que llamaremos causal $\mathcal{M}^*$
donde vive la distribución conjunta de un vector $(X, A, Y(0), Y(1))$,
$P^* \in \mathcal{M}^*$ donde $Y(0)$ e $Y(1)$ son dos variables aleatorias nuevas,
que en este momento no relacionamos con la variable $Y$ del modelo estadístico.
Como modelo causal, no es observable.

\begin{notac}
	Denotaremos $Y(0)$ e $Y(1)$ como las variables aleatorias correspondientes a un resultado
	sujeto al supuesto de que el tratamiento o intervención $A$ tenga valor $0$ o $1$,
	respectivamente, para \emph{toda} la población.

	Esto no se debe confundir con la variable estadística $Y$ condicionada al valor de $A$,
	puesto que el primero son variables en el modelo causal y el segundo lo son en el estadístico.

	Representan los resultados potenciales (\textit{Potential Outcomes}) que nos permitirán
	razonar de forma contrafactual.
\end{notac}

Veremos que $P^*$ determina $P$ en
su totalidad bajo ciertos supuestos.

\begin{notac}[Esperanza condicional]
	Denotaremos $\mu_a(x) = E[Y| X = x, A = a]$, el resultado medio de los pacientes con las covariantes $x$ que fueron
	tratados con el tratamiento $a$.
\end{notac}

\begin{definicion}[ATE causal]\cite[sección 2.1]{mci}\label{def:causalATE}
	El ATE causal, del inglés \textit{Average Treatment Effect} o Efecto Medio de Tratamiento,
	es un estimando causal definido como
	$$ \psi^*(P^*) \equiv \psi^* = \E(Y(1) - Y(0)) $$
\end{definicion}

\begin{definicion}[ATE estadístico] \label{def:ate} \cite[sección 1.3]{mci}
	Consideramos dos variables aleatorias $X$ e $Y$ y una variable aleatoria binaria $A \in \{0, 1\}$.
	Definimos el ATE estadístico como el siguiente estimando estadístico:
	$$ \psi = \E[ \E[Y | X, A = 1] - \E[Y | X, A = 0 ] ] $$
	También se puede escribir como:
	$$ \psi = \E[\mu_1(X) - \mu_0(X)] $$
\end{definicion}

\subsection{Motivación}
Dado un estimando causal $\psi^* : \mathcal{M}^* \to \R$, queremos encontrar
un estimando sobre la distribución observable que nos permita inferir sobre él,
$\psi : \mathcal{M} \to \R$, de forma que $\psi(P) = \psi^*(P^*) \; \forall P^* \in \mathcal{M}^*$.
Además, la distribución estadística se relaciona con la causal a través de
la observación: $P = \mathcal{O}(P^*)$.
Como los nombres indican, en particular querremos identificar el ATE causal con el ATE estadístico.

Visualmente, queremos que el diagrama de la derecha sea conmutativo.

\begin{minipage}{0.45\textwidth}
	\centering
	\begin{align*}
		\mathcal{O}: \mathcal{M}^* \to & \; \mathcal{M} \\
		P^* \mapsto                    & \; P
	\end{align*}
\end{minipage}
\hfill
\begin{minipage}{0.45\textwidth}
	\centering
	$$
		\begin{tikzcd}
			\mathcal{M}^* \arrow{r}{{\mathcal{O}}} \arrow[swap]{dr}{\psi^*} & \mathcal{M} \arrow{d}{\psi} \\
			& \R^d
		\end{tikzcd}
	$$
\end{minipage}

\subsection{Identificación}

El primer problema que aparece es la mayor libertad que posee el modelo causal, que resulta en
que varias distribuciones causales tienen la misma distribución correspondiente en $\mathcal{M}$.
Eso es, tenemos que las aplicaciones de estimandos causales en estimandos estadísticos,
$\pi: \left\{\mathcal{M}^* \to \R \right\} \mapsto \left\{\mathcal{M} \to \R \right\}$ no son inyectivas.

Con el objectivo de eliminar este problema se introducen las \textit{asunciones de identificación}.
Antes de ello, un ejemplo del problema.

\begin{ejemplo}\cite[sección 2.2]{mci}
	Consideramos las tripletas $(A, Y(0),Y(1))$ con distribuciones en $\mathcal{M}^*$
	degeneradas tomando como valor $(1, 0, 1)$ y $(1, -1, 1)$, respectivamente. Aplicando consistencia,
	tenemos que la distribución observable en ambos casos es la distribución degenerada de valor
	$(A, Y) = (1, 1)$.
	Este ejemplo específicamente pone de manifiesto la necesidad de la asunción de positividad,
	porque el valor de la distribución observable no dependía del valor de $Y(0)$.
\end{ejemplo}

Los supuestos de identificación que hacemos son los siguientes \cite[material suplementario S1]{luque2021} y \cite[capítulo 2.2]{mci}:

\begin{definicion}[Consistencia]
	\label{consistency}
	Una condición que vamos a imponer es la de \textbf{consistencia}, que enlaza
	las distribuciones causal y estadística:
	$$ Y = A Y(1) + (1 - A)Y(0) $$
\end{definicion}

\begin{definicion}[Intercambiabilidad condicional]
	También denominada independencia condicional o ignorabilidad. % TODO; check understanding
	$$ Y(a) \indep A | X $$
	Esto es independencia entre $Y(a)$ y $A$, bajo condicionamiento a $X$.
	% No confundir con indepedencia entre $Y(a)$ y $A | X$, denotada $ Y(a) \indep (A | X)$
	Una forma de verlo es que dado un paciente con ciertas covariantes $X$, la asignación al
	tratamiento no aporta información adicional.
	También significa que no hay más variables confusoras además de las representadas por $X$. % TODO: confusoras o simplemente covariantes?
	En particular, se obtiene
	$$
	    \E[Y(A) | A = 1, X = x] = \E[Y(A) | A = 0, X = x] = \E[Y(A) | X = x] \;\forall a \in \{0, 1\}
	$$
	    % \E[Y(a) | A = 1, X = x] = \E[Y(a) | A = 0, X = x] = \E[Y(a) | X = x] \;\forall a \in \{0, 1\}
\end{definicion}

\begin{definicion}[Positividad]
	$$ P(A | X = x) > 0 \; \forall x:  P(X = x) > 0$$
\end{definicion}

Dependiendo del autor la condición de consistencia vista en \ref{consistency}
se toma como una de ellas o no, puesto que su rol es de relacionar el modelo causal
y el modelo estadístico, y no constituye una restricción en la familia de modelos causales.

\begin{definicion}[No interferencia]\label{noInterference}
	El resultado potencial de un sujeto es independiente de la exposición o no exposición de otro sujeto al tratamiento.
	% TODO: important: eliminar la independencia entre observaciones a nivel de unidad.
	% TODO: important: "no unobserved confounding" interpretation of this:
\end{definicion}

\begin{definicion}[SUTVA]
	\textit{Stable Unit Treatment Values Assumption} (\textit{SUTVA}) combina las condiciones de
	consistencia (\ref{consistency}) y de no interferencia (\ref{noInterference}).
	Ambas condiciones se satisfacen simultáneamente bajo muestreo IID.
\end{definicion}

	trivialmente por la randomización de la asignación de los sujetos a

\begin{ejemplo}[Violación de SUTVA]
	El efecto por el cual el tratamiento a unidad afecta a otras se le llama \textit{\say{spillover}} en inglés, o \say{derrame} en castellano.
	Un posible ejemplo de experimento donde no se cumple SUTVA es el análisis % por RCT % TODO: really por RCT?
	de los efectos de una transferencia incondicional de efectivo a algunas familias en una localidad africana.
	Aquí, la transferencia de efectivo no sólo afecta a la familia que lo recibe, sino
	que también puede afectar a la localidad por el incremento en el consumo
	de esta familia.
	Este es un ejemplo del trabajo que hacen en organizaciones como GiveDirectly.
	En \cite{Haushofer2016}, seguido posteriormente por \cite{Egger2022},
	dividen el grupo de control entre \say{Control puro}, para familias no afectadas por
	el efecto de spillover, y el grupo de \say{Control con spillover}, que incluye familias
	de la misma localidad que los recipientes.
	Se puede encontrar una explicación más divulgativa de estos artículos en \cite{GiveDirectly2016}.

	Otro ejemplo es un estudio de la efectividad de una vacuna donde el muestreo se asigna por grupos (como un grupo
	de estudiantes de la misma clase)\cite{oxChapterCausal}.
\end{ejemplo}

\begin{observacion}\cite[sección 2.3]{mci}
	Dados un estimando causal $\psi$ y un estimador $\hat\psi$ del estimando estadístico identificado, tenemos
	que el error es
	$$
		\hat{\psi} - \psi^* = \hat \psi - \psi + \psi - \psi^*
	$$
	Donde a $\hat \psi - \psi $ es el error estadístico (que a su vez se puede descomponer en sesgo y varianza)
	y $\psi - \psi^*$ es la \textit{brecha causal}, o en inglés
	\textit{causal gap}.

	Si hemos identificado el estimando causal con el estimador, esta brecha causal es nula.
\end{observacion}

\begin{proposicion}[G-formula, versión no paramétrica]
	Bajo las condiciones anteriores, tomamos el siguiente estimador del ATE estadístico:
	$$
		\hat\psi = \sum_x P(X = x) \left[P(Y = 1 | A = 1, X = x) - P(Y = 1 | A = 0, X = x) \right]
	$$
\end{proposicion}

\begin{proof}\cite[material suplementario S1]{luque2021}

	Bajo las condiciones anteriores, por el Teorema de la probabilidad total (\ref{thm:probabilidad-total}) tenemos que
	$$
		P[Y(a) = 1 ] = \sum_x P[Y(a) = 1 | X = x]P[X = x]
	$$
	Aplicando la intercambiabilidad condicional, por la cual dentro de cada estrato $x$ de $X$ los predictores están
	equitativamente distribuidos entre el grupo tratado ($A = 1$)y no tratado ($A = 0$):
	$$ = \sum_c P[Y(a) = 1 | A=a,C=c]P[C=c] $$
\end{proof}

\begin{observacion}
	La G-fórmula, en su versión no paramétrica, aplica el Teorema de la Probabilidad total
	disgregando en todos los estratos de covariantes ($X$) posibles.
\end{observacion}


\begin{ejemplo}[ATE]
	Consideramos la tabla de contingencia anterior y añadimos la diferencia:
	\begin{center}
		\begin{tabular}{ C | C C | C C | C}
			i & X & A & Y(0) & Y(1) & Y(1) - Y(0) \\
			1 & 0 & 0 & 120  & 130  & 10          \\
			2 & 0 & 1 & 135  & 160  & 25          \\
			3 & 1 & 0 & 95   & 110  & 15          \\
			4 & 1 & 1 & 110  & 120  & 10
		\end{tabular}
	\end{center}
	El valor real del ATE es 15. El valor estimado por la G-fórmula es 32.5. Esta discrepancia se debe, en este caso,
	por la diferencia entre los grupos tratados y no tratados, violando la intercambiabilidad condicional:
	obsérvese que la media para $A = 0$ es $107.5$ y para $A=1$ es $140$.
	Claro está, esto no es observable, ya que hemos obtenido el valor real del ATE por el modelo causal.
\end{ejemplo}
