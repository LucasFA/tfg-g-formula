\chapter{Framework de Potential Outcomes}

\section{Intro}
Basándonos en \cite[capítulo 2]{targetedLearning}, consideremos el triplete $(X, A, Y)$ de variables aleatorias con distribución
conjunta $P$, observable en el sentido de que es posible tomar
muestras de él.
Diremos que $P$ es un elemento del modelo estadístico $\mathcal{M}$.

Consideramos ahora $(X, A, Y(0), Y(1))$, que seguirá una distribución
$P^*$, elemento del modelo causal $\mathcal{M}^*$ . Este vector no
es directamente observable, pero veremos que $P^*$ determina $P$ en
su totalidad bajo ciertos supuestos.

A $X$ lo llamamos las covariantes, a $A$ el tratamiento, y a $Y$ el resultado (\textit{outcome}, del inglés). % TODO: potentially remove outcome

\begin{ejemplo} % TODO: improve wording
	Un estudio médico puede estudiar la efectividad de un tratamiento, representado como la variable binaria $A$ según si se aplica o no,
	de pacientes de los cuales se registra la mortandad en un plazo determinado, medido en la variable binaria $Y$,
	donde tenemos en cuenta una condición binaria $X$ como puede serlo si el paciente tiene presión arterial alta o no.
\end{ejemplo}

\begin{notac}[Esperanza condicional]
	Denotaremos $\mu_a(x) = E[Y| X = x, A = a]$, el resultado medio de los pacientes con las covariantes $x$ que fueron
	tratados con el tratamiento $a$.
\end{notac}

\begin{definicion}[ATE estadístico] \label{def:ate} \cite[sección 1.3]{mci}
	Consideramos dos variables aleatorias $X$ e $Y$ y una variable aleatoria binaria $A \in \{0, 1\}$.
	Definimos un estimando, llamado el \textit{Average Treatment Effect} (estadístico) o Efecto Medio de Tratamiento como
	$$ \psi = E[ E[Y | X, A = 1] - E[Y | X, A = 0 ] ] $$
	También se puede escribir como:
	$$ \psi = E[\mu_1(X) - \mu_0(X)] $$
\end{definicion}

\begin{observacion}
	La interpretación de las siguientes condiciones reside en la necesidad de comparar los efectos causales
	dentro de un mismo grupo homogéneo, y no inter-grupos. Una ilustración de esta necesidad es presentada en \cite[sección 2, párr. 6]{rubin2005}:

	\begin{verbatim}
    One cannot compare medical compare medical costs for male
    smokers with medical costs for female nonsmokers and claim
    that this comparison is a causal effect of smoking without
    making heroic and unwarranted assumptions.
    \end{verbatim}
\end{observacion}

Los supuestos que hacemos son los siguientes \cite[material suplementario S1]{luque2021}:
\begin{enumerate}
	\item Consistencia contrafactual.
	      $$ Y = A Y(1) + (1 - A)Y(0) $$
	      Exige que la distribución del resultado coincida entre el modelo estadístico y el causal.
	\item No interferencia
	      El resultado potencial de un sujeto es independiente de la exposición o no exposición de otro sujeto al tratamiento.
	      % TODO: important: eliminar la independencia entre observaciones a nivel de unidad.
	      % TODO: important: "no unobserved confounding" interpretation of this:
	\item Intercambiabilidad condicional, independencia condicional o ignorabilidad. % TODO; check understanding
	      $$ Y(a) \indep A | X $$
	      Independencia entre $Y(a)$ y $A$ bajo condicionamiento a $X$. % TODO: check is correct interpretation (associatively)
	      % No confundir con indepedencia entre $Y(a)$ y $A | X$, denotada $ Y(a) \indep (A | X)$
	      Una forma de verlo es que dado un paciente con ciertas covariantes $X$, la asignación al
	      tratamiento no aporta información adicional.
	      También significa que no hay más variables confusoras además de las representadas por $X$.
	      En particular, se obtiene
	      $$
		      \E[Y(a) | A = 1, X = x] = \E[Y(a) | A = 0, X = x] = \E[Y(a) | X = x] \;\forall a \in \{0, 1\}
	      $$

	\item Positividad
	      $$ P(A | X = x) > 0 \; \forall x:  P(X = x) > 0$$
\end{enumerate}
\begin{definicion}[SUTVA]
	\textit{Stable Unit Treatment Values Assumption} (\textit{SUTVA}) combina las condiciones de consistencia y de no interferencia.
	Ambas condiciones se satisfacen simultáneamente bajo muestreo iid.
\end{definicion}

\begin{observacion}\cite[material suplementario S1]{luque2021}
	En estudios randomizados se mantiene la intercambiabilidad condicional
	trivialmente por la randomización de la asignación de los sujetos a
	los grupos de control y de tratamiento.

\end{observacion}

\begin{ejemplo}[Violación de SUTVA]
	El efecto por el cual el tratamiento a unidad afecta a otras se le llama \textit{\say{spillover}} en inglés, o \say{derrame} en castellano.
	Un posible ejemplo de experimento donde no se cumple SUTVA es el análisis % por RCT % TODO: really por RCT?
	de los efectos de una transferencia incondicional de efectivo a algunas familias en una localidad africana.
	Aquí, la transferencia de efectivo no sólo afecta a la familia que lo recibe, sino
	que también puede afectar a la localidad por el incremento en el consumo
	de esta familia.
	Este es un ejemplo del trabajo que hacen en organizaciones como GiveDirectly.
	En \cite{Haushofer2016}, seguido posteriormente por \cite{Egger2022},
	dividen el grupo de control entre \say{Control puro}, para familias no afectadas por
	el efecto de spillover, y el grupo de \say{Control con spillover}, que incluye familias
	de la misma localidad que los recipientes.
	Se puede encontrar una explicación más divulgativa de estos artículos en \cite{GiveDirectly2016}.

	Otro ejemplo es un estudio de la efectividad de una vacuna donde el muestreo se asigna por grupos (como un grupo
	de estudiantes de la misma clase)\cite{oxChapterCausal}.
\end{ejemplo}

	$$ Y = A Y(1) + (1 - A)Y(0) $$
	y $(X, A)$ sigue la misma distribución entre el modelo causal y el modelo estadístico:
	$$ P^*(X = x) = P(X = x) \;\forall x $$
	$$ P^*(A = a | X = x) = P(A = a | X = x) \;\forall x, a$$

	$$ P(Y, A, X) = P^*([AY(1) + (1 - A)Y(0)] \; |\, A, X) P^*(A | X) P^*(X) $$
\end{proposicion}

\begin{proposicion}[G-formula]
	Bajo las condiciones anteriores, el ATE (estadístico) puede estimarse por:
	$$
		\hat\psi = \sum_x P(X = x) \left[P(Y = 1 | A = 1, X = x) - P(Y = 1 | A = 0, X = x) \right]
	$$
\end{proposicion}

\begin{proof}\cite[material suplementario S1]{luque2021}

	Bajo las condiciones anteriores, por el Teorema de la probabilidad total (\ref{thm:probabilidad-total}) tenemos que
	$$
		P[Y(a) = 1 ] = \sum_x P[Y(a) = 1 | X = x]P[X = x]
	$$
	Aplicando la intercambiabilidad condicional, por la cual dentro de cada estrato $x$ de $X$ los predictores están
	equitativamente distribuidos entre el grupo tratado ($A = 1$)y no tratado ($A = 0$):
	$$ = \sum_c P[Y(a) = 1 | A=a,C=c]P[C=c] $$
\end{proof}
