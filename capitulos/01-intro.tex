\section{Potential outcomes}
Consideremos el triplete $(X, A, Y)$ de variables aleatorias con distribución
conjunta $P$. Este triplete es completamente observable: es posible tomar
muestras de él, y diremos que $P$ es un elemento del modelo estadístico $\mathcal{M}$

Consideramos ahora $(X, A, Y(0), Y(1))$, que seguirá una distribución
$P^*$, que diremos es un elemeneto del modelo causal $\mathcal{M}^*$ . Este vector no es directamente observable, pero veremos que $P^*$ determina $P$ en
su totalidad:

\begin{proposicion}
	Si el modelo causal y el estadístico son consistentes:
	$$ Y = A Y(1) + (1 - A)Y(0) $$
	y $(X, A)$ sigue la misma distribución entre el modelo causal y el modelo estadístico:
	$$ P^*(X = x) = P(X = x) \forall x $$
	$$ P^*(A = a | X = x) = P(A = a | X = x) \forall x, a$$

	Entonces el modelo causal determina totalmente el estadístico:
	$$ P(Y, A, X) = P(Y | A, X) P^*(A | X) P^*(X) $$
\end{proposicion}

\begin{proof} \cite{schuler21}
Tenemos que:
$$ P^*(X, A, Y(0), Y(1)) = P^*(Y(0), Y(1) | X, A) P^*(A | X) P^*(X)
$$
Tomando
$$ P(Y = y | A, X) = P(AY(1) + (1-A)Y(0) = y | A, X) $$
Entonces
$$ P(Y, A, X) = P(Y | A, X) P(A | X) P(X) $$
\end{proof}

\subsection{Diferencia de riesgo}

\begin{definicion}[Diferencia de riesgo] \label{def:riskdiff}
	Dado un tratamiento, definimos la Diferencia de Riesgo como
	$$ RD = \frac{E_A}{E} - \frac{C_A}{C} $$
	donde $E_A$ es el número de casos
\end{definicion}
