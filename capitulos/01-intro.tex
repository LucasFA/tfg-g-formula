\begin{definicion}\label{def:ate}
	Consideramos dos variables aleatorias $X$ e $Y$ y una variable aleatoria binaria $A \in \{0, 1\}$.
	Definimos el _Average Treatment Effect_ como
	$$  E[ E[Y | X, A = 1] - E[Y | X, A = 0 ] ] $$
	Llamaremos a $X$ las covariantes, $A$ el tratamiento e $Y$ el resultado.
	%TODO: check "covariantes" is a proper translation
\end{definicion}

\begin{definicion} Diferencia de riesgo \label{def:riskdiff}
	Dado un tratamiento y un gru
	$$ RD = \frac{E_A}{E} - \frac{C_A}{C} $$
	donde $E_A$ es el número de casos
\end{definicion}
